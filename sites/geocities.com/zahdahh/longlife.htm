طويل العمر
 عبد الوهاب زاهده 
ماذا        إذا       ماتَ       iiالزعيم
يا                                iiويلنا
من        ذا       سيخذلُ       iiخيلَنا
من         ذا        يعوّدُ        iiخدّنا
فنِّ                              iiاللطيم
***
ستكونُ     كارثةٌ    وتحدُثُ    iiزلزله
فلقد             كتبنا             iiإسمهُ
في    كلِّ    زاويةٍ    وفوقَ   iiالمزبلهْ
ولقد            نقشنا            iiرسمَهُ
في   ظهرِ   عملتِنا   وصدرِ  iiالمقصلهْ
يا                                iiويلنا
من       ذا       يُصرّفُ       iiأمرنا
من       ذا       سيقصُمُ       ظهرنا
ولمن           نجيّرُ           iiصوتنا
عاشَ                           iiالعظيم
***
قالوا   لنا   من   بولِهِ   يشفى  iiالسقيم
وعلى    رذاذِ    بُصاقِهِ    تلدُ   العقيم
وهو          المسببُ         iiوالسبب
وهو       الندى      وهو      iiاللهب
لولاهُ        ما       كان       iiالعرب
فوقَ                             iiالأديم
يا                                iiويلنا
ماذا        إذا       عنا       iiإحتجب
من        ذا        سيفرُمُ       iiلحمنا
من        ذا        يُهشّمُ       iiعظمنا
من         ذا        يُرَمّلُ        iiأمّنا
ويُيَتّمُ           الطفلَ           iiالفطيم
***
لا   لن   يكونَ   شبيهُه  في  iiالعالمين
فلقد     خصانا    واحتوانا    iiأجمعين
من       ظلمِهِ       نجري       iiإليه
ونخافُ       لو       ندعو      iiعليه
نجثو            لنلثمَ           iiراحتيه
فإذا            تواضعَ            iiسبّنا
وإذا             تمادى            iiكبّنا
للريجِ   تذرونا   كما   يُذرى   iiالهشيم
يا                                iiويلنا
ممن  سيأتي  بعدَهُ أيكونُ إبليسُ iiالرجيم
***
كم    ذا   يكلّفنا   الخليفةُ   من   iiنفاقْ
سيكونُ       أمرٌ       لا       iiيطاق
سنجر رغم أنوفنا من كل بيت أو زقاقْ
حتى            نجرّّعَ            iiمُرّنا
حتى            نجدد           iiعهَرنا
يا                                iiويلنا
فالعبدُ    يخصى    مرةً    لا   iiمرتين
لكنّهُ     قدرٌ     علينا     بل    iiودَين
أن    نبقى    دوما    بينَ    ..   بَين
لا   نحنُ   نحسبُ   سادةً  أو  iiكالعبيد
مأساتنا    أن    الكبير    مع    iiالوليد
مستنسخون         عن         الزعيم
عاشَ                           iiالزعيم
