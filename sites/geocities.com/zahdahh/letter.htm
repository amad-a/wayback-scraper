للمغتربين فقط
 عبد الوهاب زاهده 
مِـن غـيـرِ ما طبٍّ ولا iiتِرياقِ
أنـا  عـلّـتي أنّي من iiالعُشّاقِ
مـن  ذا يُداويني ليهداَ iiخاطري
وأنـامَ مـلءَ الـعينِ والأحداقِ
إنـي عشقتُ العشقَ حتى iiنفسَهُ
من  ذا يضيفُ الشوقَ iiللأشواقِ
قـالـوا  الـليالي قد تبَدّدُ iiحبَّنا
فـالـحبُّ قد يخبو فليسَ iiبباقي
إلا  هـوايَ فـغـيرُ مجدٍ iiغفلُهُ
تـحـيـا عليهِ حشائشُ iiالخفّاقِ
كـلُّ  الأحـبّـةِ قد يحَدّدُ iiنُزلُهم
إلا حـبـيـبي .. نُزْلُهُ iiأعماقي
ألقاهُ في الصالونِ فوقَ iiمقاعدي
ألـقـاهُ  مـلتصقاً مع iiالمغلاقِ
كـالـطـفلِ يسحبني إلى iiألعابِهِ
ويَـشُـدّنـي  مُـتعلّقاً iiبخِناقي
أنّـى  جـلـستُ يجيئني iiمتلفّعاً
بـعـبـاءةِ الـزيتونِ iiوالدرّاقِ
أنـا يـا أبـي أشتاقُ كلَّ iiدقيقةٍ
لا شـيء يشغَلُني على iiالإطلاقِ
إلا الـحـنينُ فكم يُعَذبني iiالنَوى
أشـتـاقُ يـا أمي إلى iiالأطباقِ
هـلا سـألتِ الخالَ كي يبقي لنا
بـعـضاً من الألبانِ في iiالدَرّاقِ
قـولـي لأمِّ مـحمّدٍ لو سافرت
يـوماً  تبيعُ الخسَّ في الأسواقِ
تـأتـي إلـيَّ من الخليلِ بكرمةٍ
مـن  سـوقِ غزّةَ فلفلٍ iiحرّاقِ
مـن بابِ خانِ الزيتِ سلِّ iiفواكهٍ
ومـن  الـشمالِ مُسّخنٍ iiبرُقاقِ
والحاجُ .. كيفَ الحاجُّ ما iiأخبارُهُ
أتُـراهُ والـمـختارُ دونَ iiوفاقِ
وهل المضافةُ مثلُ سابقِ عهدها
يـأتـي  إلـيها أعزّتي iiورفاقي
أنـتـم مـعـي بتأمُلي وتخيُّلي
إنّ  الـتـخـيّـلَ نعمةُ iiالخلاقِ
فـإذا  قرأتُ رأيتُكم في iiأسطري
وإذا  كـتـبـتُ فأنتمو iiأوراقي
تـأتـونَ في المرآةِ كلَّ صبيحةٍ
فـي البيتِ بل حتى لدى iiحلاقي
يـا  ليتكم أبقيتموني في iiالحِمى
يـا  لـيـتكم شدّدتموا iiبوثاقي
يـا ليتها قدمي .. تفتتَ iiرُسغُها
يـا  ليتها لو لم تكن هي iiساقي
يـا  ليتها كُسرتْ ولا هذا القذى
فـي  العينِ من طوزٍٍ بكلِّ iiزقاقِ
يـا  ليتني لم أغتربْ عن iiديرتي
هل  من فتىً قد ماتَ من iiإملاقِ
