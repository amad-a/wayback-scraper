في بيتنا شيطان
 شعر عبد الوهاب زاهده 
في     بيتنا     iiشيطانْ
بالصوتِ       iiوالألوانْ
ما      أن      iiنلامسَهُ
حتى     يفيقَ    iiالجانْ
فنرى    على   iiالشاشهْ
الجندَ           iiوالباشا
وجلالةَ         السلطان
*****
وكما     هيَ     iiالعادهْ
تمتدُّ             سِجّادهْ
من     فوقها    iiيمشي
ليودّعَ            iiالقادهْ
والزلمَ         iiوالأعيانْ
وقناصلَ         iiالبلدانْ
****
ويواصلُ         السفرا
في    صحبَةِ    iiالكامِرا
لنظلَّ    في    iiالصورهْ
ونتابعُ           السيرهْ
فلأجلِنا           iiيشقى
يتحمّلُ          iiالطيرانْ
فالصبحُ     في    بُرقهْ
والعصرُ   في   iiاليونانْ
يقضي الضحى في نيسْ
وينامُ     في    iiباريسْ
يتَسوّقُ          iiالعطرا
والفسقَ        iiوالفُجرا
ويجيءُ    في   النشرهْ
من      أجلنا     iiسافرْ
ولأجلنا          iiحاضرْ
وكلامُهُ            ثوره
في    العلمِ    iiوالإيمانْ
ومبادىء       iiالطغيانْ
*****
تأتي             iiحقائبُهُ
ملأى     من    iiالرحلهْ
نخشى          iiنُحاسبُهُ
وتُعيّدُ            iiالدولهْ
نصّطَفُ   في  iiالطرقاتْ
وتُرفرِفُ        iiالراياتْ
وتُزغردُ        النسوانْ
ويُصفّقُ         iiالغلمانْ
****

من      أجلنا     iiبكُرهْ
