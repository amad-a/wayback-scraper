شمسُ أيار...

 

 

ظمأى له ُ في قيدها الأوطانُ

 

أيار ُعادَ ، وشمســهُ لبنانُ

ترنو إليه  وعـيشها الأحزانُ

 

لمعت ْ بشارةُ نصره ِأملا لها

دوَّت به ِ الأجراسُ والآذانُ

 

مرحى لنصرٍ فيه عزةُ مسلم ٍ

يشقىبه ِالطاغوتُ والعدوانُ

 

أبقى طنينا في المسامع لم يزل

شفيت وغصَّ بقرحهِ الطغيانُ

 

أسعدت َفينا أمة ً وصدورُها

أخرى ، لمجد ِ الأمة العنوانُ

 

فإذا الجنوبُ ، كأنَّ فيه قبلةً

شجنٌ  تحرِّكهُ ُ بهـا الأزمانُ

 

تهفو إليه قلوبُ ما انفكت لها

والقدسُ في أخرىلهاهميانُ

 

يرنو بعينِ الفخرِ صوبَ حُماتهِ

وعدٌ ، تحنُّ لصبحهِ الفرسـانُ

 

أشرقتَ حزبَ الله ِ نصراً بعضهُ

واللهُ غالــبُ والخطى قرآنُ

 

أهديتَهُ الإسلام َحراً كيفَ لا

بدمٍ توحَّدَ وارتقى البنيـانُ

 

يوما زرعنا بذرةً قد أنبتت ْ

شمسُ الفداءِ ، هوية ٌ ورهانُ

 

ياكربلاءَ العصرِ قد سطعت بهِ

قولُ ابنِ طالبَ قالـهُ عثمانُ

 

قدَّرتَ أنَّ خيارَهُ من ارثـه ِ

قالَ الإمـام ُ وأمنَّت طهرانُ

 

أكرمْ بها من عهدةٍ ووصيَّةٍ

مجدُ الفداء ِ وبندقيَّةُ  مؤمن ٍ بابٌ  إلى  الفردوس ِ وهي  جَنـانُ

فكراً وآسادا ً لها الميدانُ

 

من قاسيونَ لبانةٌ قد أطلعت ْ

درعٌ وهـمْ للصامدينَ سنانُ

 

أبناءُ جلِّـقَ للعروبة ِ جُنَّــةٌ

هلا نظرتَ ركـابهم مروانُ

 

ما ضرّهم من أمسهم ما وُرثوا

يعدُ الجليلَ ، وصبحهُ الجولانُ

 

نُبئِّتُ أنَّ البِشرَ فيها قد غدا

في أرضِ مصرَ يقوَّمِ الميزانُ

 

يممْ بوجهكَ شطرَ بغدادٍ وقمْ

إن الشقاقَ جريمــةٌ وهوانُ

 

من لم يقم بالنُّصحِ قامَ بغيره ِ

فاليـوم َ لا عذر ٌ ولا غفـران ُ

 

لو كانَ فينا مَن تعلل بالأسى

أو عادَ فينا المجدُ والبرهانُ

 

إما درسنا وانتهت آثــارُنا

والحــقُّ لا خورٌ ولا خـذلانُ

 

من يرتقي للعزِّ يكبرُ حقَّـهُ

يومَ اللقاءِ  يشدُّها الإيمــانُ

 

واللهُ باركَ في الصفوفِ يظلُّها

بالنورِ سجِّل ْ انـَّهُ الإعــلانُ

 

يا كاتبَ التاريخِ سجِّل ْ يومنـا

بأفولِ عصرٍ  سـادهُ  الشيطانُ

 

زمنُ انتفاضةُ قدسنا قدْ آذنت

وينيرُ عهـدٌ  للهدى ظمـآنُ

 

هو عهدُ طغيانٍ سيقضي حسرةً

سقطت منابرُ وانثنت جدرانُ

 

المسجدُ الأقصى يبيتُ مناجيـاً

أزجى الجيوشَ  وقلبهُ حيرانُ

 

يتلفتُ اليمنى لعــلَّ نداءه ُ

وردوا المنايا والردى آشطانُ

 

فإذا بهم فتيانُ أكنافٍ وقد

كالقطر ِأولُ غيثـهِ الطوفـانُ

 

رضعوا حليبَ المؤمناتِ وهم بهِ

بدرٌ ويومُ القدسِ ذا صنوانُ

 

ياأيها الوطنُ السليبُ لك العلى

فاليومَ بذلٌ والرحى ألــوانُ

 

قد أقسموا واللهُ برَّ يمينهــم

عزَّ المتاع ُ و أنفسَ الانسـانُ

 

يفدونَ بيت الله ِ ما ضنُّوا بما

فهي الكماةُ  تجرُّها الأبدانُ

 

أرواحهم قدنازعتهم في الفدا

بشرى يحقق ُ نورها  نيسـانُ

 

بوركتَ يا جيلَ الشهادةِ انها

 

 

 

 

 

                ايمن اللبدي
