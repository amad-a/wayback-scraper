للصَّمت ِ أكثرُ من لغة ْ

للصمت ِ حاشيتان ِ من وحش ِ الغبار ْ

للصمتِ مائدةُ العشاءِ المستعيدةُ فسقَها

للصمتِ دولابُ الخروجِ من الفسيلةِ والشبَق ْ

للصمتِ سردابُ القميصِ وإفكُ صاحبةِ الطهارةْ

للصمتِ مسمارُ الصليبِ على خلاصِ الاستعارةْ

للصمتِ قافلةُ الخطايا والحبَق ْ

للصمتِ سكّينُ العبارةْ

للصوت ِ نافذتان ِ في وجهِ  الخيار ْ

للصمتِ أكثرُ من لغةْ........

 

             *****

للصمتِ واحةُ إمريء القيسِ المعذَبِ والفحيحْ

وله ُاحتباسُ ابنِ الوليدِ على رضابِ الراقصاتْ

وله ُ الجهاتْ

ليسَ الموشَّحُ قرطبة ْ

ليسَ الذي أبقاهُ في يدهِ ابنُ زيدونَ الذبيحْ

ليسَ الذي أعطاهُ من فمه ِ المرقِّشُ واحتفى

بالموتِ فيهِ أبو العلاء ْ

خاِطبْ شفاهَ الليلِ والدارَ الحبيسةَ والأرقْ

وارقصْ على أنغامِ كثبانِ الرمالِ الراحلة ْ

وتعاهدِ الريحانَ في دارِ الطريدةِ والورقْ

ليسَ المخدّرُ ثعلبةْ

أبق ِ الذي تحتَ الوسائدِ في الدواة ْ

لا تكتبِ اليومَ الخطابْ

لا ترتدي قلقَ السِّفاح ِ المستفيض ِ مخاضُه ُ

واجمعْ كتابكَ من جديد ْ

لا همَّ إلا ما تريد ْ

فقدِ استراحَ العابرونَ على الهضابْ

دعهم على لغةِ البكاءِ المستريحْ

دعهم هناك ْ

للصمتِ أكثرُ من لغة ْ......

 

             *****

للصمتِ باحةُ من يعذِّبهُ القمرْ

من ْ يكتوي فيهِ الذهولُ على شبابيكِ الأسى

من ْ يستزيدُ جلالةَ الليلِ المسافرِ في الشواطيءِ والغيوم ْ

أو في الجداولِ والسنابلِ والقزحْ

قيثارةٌ أدمت يديهِ من الهوى

عصفورةٌ سلَّت بجفنيها السفرْ

وتعجَّلت تغريبةً نحو البعيدِ من التخوم ْ

لا مرَّةً قد أخلفَ الوعدَ ولا

هو أسلمَ الروحَ التي في خافقيهِ دونَ ملاكِها

كلا وحاشا لم يكذِّبها الخبرْ

ومضى يترجمُ كلَّ ما وقفت بها يوماً قوافلُ متعبة ْ

هربت من المدنِ المضلَّلع ِ وجهُها

وتقاطرت بينَ السلامة ِ والخطرْ

مازالَ يروي سندبادْ

مازال يحترفُ الحيادْ

يبكي هواهُ المختَطفْ

ويعودُ يغلقُ كلََ يومٍ بابَ قلعتهِ المسربلِ بالوجع ْ

ويقودُ دولةَ ذاته الغنّاءَ في عصرِ الزجاج ِ

وعصرِ فاتحةِ البشرْ

شكواهُ إنْ ذبِلتْ تقطَّع إرثُها بين اللصوص ْ

أواهُ من شكواهْ

فلتسترح ْ شفتاهْ

للصمتِ أكثرُ من لغة ْ......

 

             *****

للصمتِ حيطانُ الجنون ْ

ولهُ انتظارُ الأسئلةْ

ولهُ اختراعاتُ المسائلِ والتَّبعثرُ في إدعاءِ المعرفةْ

ولهُ الوقوفْ

ما بينَ حلقاتِ العبثْ

وتقوقع ِ القبرِ المبالغ ِ في الخلايا المرسلةْ

ولهُ الذينَ تبرعوا أن ينحتوا مدناً من اللاشيءِ قبلَ رحيلهم ْ

واستغرقوا في" اللا" وخاطوا نفرةَ الذهنِ المحاصر ِ بالولوجْ

وتذرعوا بالمعضلةْ

ما أكذبَ استدعاءَ خَلْقٍ من زردْ

لم يقتلوا  بعدَ التلاوةِ ربَّهم ْ

بل إنما قتلوهُ قبلَ رحيل ِ "كامو" بالصواعْ

وبكوهُ تحتَ المقصلةْ

لا إرثَ للمقتول ِ بينَ الذهنِ والجسدِ المصفَّح ِ بالثقوبْ

لا توبةً تجدي إذا قامَ الحبيسْ

وتفرَّقت بينَ القبائلِ دودةُ اللغزِ المفبركِ

واشتهت شيطانها حزمُ الذنوبْ

لن تستردَّ ضبابها إن أفلتتهُ من جديدْ

فليستمرَّ الاحتقانْ

وليدِّعي ما شاءَ من شبق ِ احتفالِ الأحجيةْ

وتسلُّق ِ الأفلاك ِ خلفَ الأزمنةْ

والانتظارْ

لا تيقظوهْ

ودعوهُ يبحثُ سرََ " كافكا " المعجزة ْ

للصمتِ أكثرُ من لغة ْ....

             *****

للصمتِ نهدٌ مستفزٌ في العراءْ

محمولةٌ أنخابهُ بينَ القرنفلِ والمجونْ

مسبيةٌ أختامهُ في رشفةِ الخبرِ المعطَّرِ بالوترْ

والمحتفي بالياسمينَ وثورةِ الحنّاءْ

ولرحلةِ الغجريِّ فوقَ الريحِ بينَ الماءْ

وتسلسلِ الأفخاذ ِ والأقدام ِ والأفراح ْ

ولهُ الوسائدُ والمرايا والبخورْ

ولهُ المشابكُ والعطورْ

ولهُ القصورْ

ولهُ افتتاحُ الاشتهاءْ

ولهُ احتفالُ المتعبينَ على الجفونْ

ولهُ جيوشُ الراقصينَ على النحورْ

من يستطيعُ نزاعّهم ْ ؟

من يستطيعُ  خداعهم ْ ؟

والويلُ للعرافِ والسِّحرِ المحضَّرِ واللحونْ

قد حاكموه ُ وقرروا

مفقودُ أولَّ  أمرهِ و إذا بغى مقهورْ

حرّاسُ قلعتهم ْ

عشّاقُ خيمتهم ْ

نسلٌ يجدِّدُ كلَّ يوم ٍ دورةَ الفزع ِ المخضَّب ِ بالجسدْ

فدعوهُ حيثُ يموء ْ

للصمت ِ أكثرُ من لغة ْ......

             *****

 

للصمت ِ خطبةُ ماردٍ بينَ الدخانِ وظلِّه ِ

يخطو فتلحقهُ الرياحُ اللاهثة ْ

وإذا توعَّدَ ضمَّت ِ الأرضُ المفاصلَ واشتهت ْ

لو أنْ تصفَّدَ راكعةْ

من أينَ يأتيها الفرارْ ؟

والبابُ مسدودٌ وتُلقي فسحةُ التوت ِ البقيَّة َ متعبة ْ

قد أدركتْ عينَ المصيرِ وأسدلت أجفانها

هذا حصارْ

يا ويلها من سطوةِ الجنيِّ والموتِ الصَّخبْ

وغضب ْ... غضب ْ

تمشي إليها الأخيلةْ

وتمورُ في حيطانها الصورُ الهلاكْ

من يفتديها ياترى ؟

كلا وحاشا لا أحد ْ

قد أسلمتهُ أمرها

هذا انتحارْ

فإذا الضجيجُ قد تبَّخرَ فجأةً

وإذا الدخانُ لم يعدْ من حولهِ

وإذا هو القِدرُ المغلَّقُ بابُهُ بالنّملِ والصَّدأ البطيء ْ

ها قد نجتْ

للصمتِ أكثرُ من لغة ْ.....

             *****

للصمتِ سوقُ المتعبين َ الماكرين ْ

لم يستطيعوا أن يكونوا في المزاد ْ

لا صيفَ يرجى

أو شتاءً خلفَها حتى تقررَّ وجهَها الآنَ قريشْ

والبيتُ يحميه الذي يدهُ الزنادْ

فيمَ العتابْ ؟

مازالَ وقتٌ كي يكرِّرَ صاحبُ الفيل ِ السؤالْ

فلتخسأ الثاراتُ ولتصمتْ إلى الأبدِ المديدِ كلُّ داحسَ غيرِنا

هذا اختراعٌ بيننا

سدّوا الطريقَ إلى الرجوعْ

إن أغلقَ السوقُ نعودُ لنشتري ونبيعَ من خبر ِ الرمادْ

والربحُ أوفرُ في الخريفِ وحيثُ تأتينا الزوابعُ بالبقايا والضلوعْ

فلنكتفي بالرمل ِ والشِّعرِ الدروعْ

لم نعترفْ يوماً بخيمةِ من تجرَّأ خارجَ السربِ المنشَّر في الجبالْ

كلا ولن نسمحْ بأن ترثَ الروافدُ والشواطيءُ ما فقدنا من أثرْ

للصمتِ أكثرُ من لغةْ......

             *****

للصمتِ أكثرُ من لغة ْ

للصوتِ نافذتانِ في وجهِ الخيارْ

للصوتِ شاهدتانِ من نمطِ الروايةْ

ماذا تبقى من غطاء ْ؟

ماذا تبقى في السماءْ ؟

فلتحتفي تلكَ الفواصلُ ما أرادت إنها

لن تختفي بينَ السطورِ المفرداتِ ولن تركِّبَ جلدها خلفَ الحكايةْ

ولن تخلِّدَها الخيانةْ

لا بد من صوتٍ يطهِرُ باحةَ الصمتِ الصفيق ْ

ويستعيدُ المطرقةْ

ويصدرُ الحكمَ الذي طالَ انتظارُ خطابهِ

واختارَ فيما اختارَ أن تبقى البطولةُ في الفضاءِ مداولةْ

لا بدَّ من حُكْم ٍ يكاثرُ شاشةَ الأملِ العجيبة ِ في النهارْ

ويستعيدُ الوجهَ من خيط ِ الوقوف ِويرتوي

ما قد توسَّدَ في النهايةْ

لا ..لا فرارْ

لا بد أن يلدَ القرارْ

لا بلدَ أن يلدَ القرارْ..........

 

 

أيمن اللبدي

9/2/2003
