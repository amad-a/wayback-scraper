المجلس الوطني الفلسطيني

دورة الجزائر- نوفمبر 1989

 

             دولة ونشيد .....

                                  

والفجر ُ لاح َ  بدولـة ٍ و نشيد ِ

 نثروا المدى وردا ً وخفق وريد ِ

بشرى التحرُّرِ درب ُكل ِّعنيدِ

نورٌ إذا شق َّ الفضاء َ تهللت ْ

بالسعد ِ والخيرات ِ والتمجيد ِ

مالي أرى الدنيا ترتـِّـل آية ً ؟

أملا ً وبات الكون ُ جـدََّ  سعيد ِ

لبست من الأفراح ِ أبهى حلة ٍ

ذا الشعب ُطوَّقها بعرسِ ِشهيدِ

ألفيت َ في كبد ِ السماء ِ مواكبا ً

والدمع ُ رق َّ لفارس ٍ و خريد ِ

بين الصدور ِ لواعج ٌ مبثوثة ٌ

وقف َ الزمان ُ تحية َ التوليــد ِ

لما تلا الوطن ُ الإرادة َ عندها

كانت  لشعب ٍ ثائر ٍ و مجيــد ِ

وتلألأت في الحقِّ خير ُمسيرةٍ

علم ٌ ، وعزته ُ من التوحيـــد ِ

مرحى لشعب ٍ لم يهن ْ وكفاحه ُ

ألقت  بهم في وحشة ِ التسهيد ِ

الله ُ أكبرُ ، زلزلت بنيــــانــهم ْ

ويل ٌ لهم  من غضبة ٍ ووعيد ِ

قد بات  كل ُّ الغادرين َ أذلــة ً

ما  بين َ مقترب ٍ وبين َبعيــد ِ

طال َ انتظار ُ اليوم ِ في أيامنا

والشعب ُ ممتحن  ٌوفي تشريد ِ

حتى ظننا العمر َيمضي كلـــه ُ

وطني  رداء العزة ِ المنشود ِ

بل قد تحقق خير ُحلم ٍ وارتدى

بين الشعوب ِمثال َخيرِصمود ِ

هي وقفة الأحرار ِكم قد سجلت

فيه الدماءُ ،جلالة َ التجسيـــد ِ

إن الفداءَ  لنا غرام ، ٌ جسَّدت

ترقى إلى العلياء ِ والتخليـــد ِ

شعب ٌ توحد في الكفاح  وأمة ٌ

هي صولة ُ التخليق ِ والتجديد ِ

حرب ُالحجارة ِليس ضعفا إنما

ما بين َ جنرال ٍ وبين َ عميــد ِ

أبطالها  أترابهم  غـر ٌ وهــم  

والحرب َ حفوها  بكــل ِّ فريد ِ

رضعوا لبان َ المجد ِمن فلواتها

بدم ِ الكرامةِ فهو خير ُ رصيد ِ

تلك السواعدُ قد بنت هرماً لنا

كسروا القيود َبقبضة ِالصنديد ِ

داسوا بأقدام ِ الإرادة ِ خيبـــة ً

حراسه ُ دمنا  ، ولا  بوحيـــد ِ

المسجد ُ الأقصى فليس َ بمفرد ٍ

عهدَ السماء ِ وارث َخير ِجدود ِ

نحن ُ الذين َبنا الرباط ُ فمرحبا

شوك ٌ بحلق ِالغدرِ،غيظ ُحسود ِ

صخر ٌعزائمنا ، وموتٌ حربنا

فاهنأ بها يا شعب ُ ياابنَ الصيدِ

هي وقفة ُ الأبطال ِملحمة ُالعلا

فإذا الردى  وهم ٌ وقيد ُ عبيـــد ِ

من عهد ِأحمدَ لم نزل ْ فرسانها

من غير تفريط  ٍ، وبيع ِ تليـــد ِ

غصنُ  السلام ِ وبندقيةُ  ثــائر ٍ

أو يصدح اللهـــب ُ ونارُ حـديد ِ

إما الحقوق ُ على تراب ٍ كامل ٍ

من بحرها للنهر ِ بيعة ُ جيـــد ِ

هذي فلسطين ُ التي هي َ قبلة ٌ

وقف ٌ على الإسلام ِدونَ يهود ِ

لسنا  نفـرط ُ في  ثراها  إنها

في الدرب ِ للتحرير ِ والتشييد ِ

الـيوم َ نعلنــها  ولكن ْ خطوة ٌ

لم تفهم  الدنيا  خطاب َرشيــد ِ

فاحمل ْ سلاحك انه أمل ٌ إذا

وترفرف  ُ الأعلام ُ يوم َ  العيد ِ

حتى يعود َ العدل ُ فهو َ مغيَّب ٌ

وان فتح ٌ تلاها كان َبيت َقصيد ِ

هي بيعة ُ الرضوان ِفي الماضي
