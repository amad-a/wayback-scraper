صرخة مسلم .....            

 

 

 

 

 

لهفـــي على أيامــــه النضراتِ

لهفـي على الإسلام والحرمــات ِ

والديــن والأمجـــاد والبركـــاتِ

تاجِ ِ الخلافة والحضـارة والعلا

والـعـــلم والعمران والملكــــاتِ

عهد ِ الكرامة والرجولة والفـــدا

فـيمـــا أراهُ تلاحـــق الأزمــاتِ

واغوث أمة أحـــمدِِ ِ قد هدَّهــــا

حرزاً فصـار لضيعـــة ٍ وشتاتِ

يا أمة الإسلام ِ كان حياضُـــــها

في وابـل التمزيــق والعــثراتِ

وتفتّت في رأيـــها فتنـــــاثرت ْ

أحيــــاءَ رغم عظامها النخراتِ

هلا ذكرتم في التراب أحبـــــة ً

بكت السمـــاء ُ لهولهـا العبراتِ

أجدادكم حملوا الأمانـة َ حمـــلة ً

شهــــداء ُ في عدن من الجنـاتِ

صانوا وابـلوا في المواقع كلـها

 

 

   كانت لنا الراياتُ تخفــقُ في المدى وضــــاءة ً وعزيــــزةَ الجنبــــاتِ

 

 

وشـــهادةٍٍ  إنْ عزّ  حــرُّ حــيـاة

كان الحرام َ حماه ُ ،هيبة َ ظافرٍ

بين َالأوابــدِ صار فضــلَ فتاتِ

مالي أراه ُاليوم نهبا في الورى

وســــواه ُ ينهشُ َمجْمعَ الربلاتِ

هذا يقطّع ُ بالنواجــذِ قصعـــة ً

ودمـــاً مهرناه ُ، علــى الغزواتِ

أممٌ تخاطف ُ ما ورثنــا تالـــداً        

كالظــــئر ِ تدفع مبلغ الحلمــاتِ

ظلت جيوش ُ الفاتحين تحوطه ُ

حتى كأن ّ القلـــب َ رهنُ سباتِ

ما بالنـــا قد ران َ فينــا وازع ٌ

لــمكيـدةٍِ أو مـهلك الـــنزواتِ

ننسى ويذكرُ غيرُنا ما يشتهـي ؟

مثل الغثــاءِ يسيل ُ في القنواتِ

إنا ملئنــا البــرّ لكـــن ْ كثــرة ٌ

لــيكون َ إفناء ً وشطبَ سمـاتِ

هذا مصير بائــــس ٌ ما ضرّه ُ

 

 

     لولا سبائب ُ في الهدايةِ  أُشرعت ْ لعلامة ِ التوحيــد حصن ُ ثقاة ِ

 

 

والنصرُ يحدوها مدى السنواتِ

يا راية التوحيدِ تخفق في المدى

إن ْ زاغ َ أقـــوام ٌ لـــوهن ِ قناةِ

هذا هو المنهــاج ُ أوثق ُ قائــدا

دمتـــمْ ودامـــتْ هيئة العزماتِ

هي هيئة والذئب ُ يبكي صـيدهُ

وتهتـّــكُ الأعراضَ بـالعشراتِ

الصرب ُ تذبح ُ منْ يوحـدُ ملـة ً

تشكــــو بنـوح ٍ يقطعُ الفلذاتِ

المسلماتُ القانتاتُ خـــدورُهن ّْ

نـــامتْ عيونُ المكرِ والظلماتِ

يستولدونَ حرائراً غصباً  فــلا

كشفتْ نيوبَ الزيفِ في الدعواتِ

همجية ٌ هبــطت لدرك ٍ  اسفلِ ٍ

لا مطلبــــــا لمدارج اللـــــذاتِ

حربٌ علىالإسلام حربُ ضغينة

تبت يد الهندوس في الحسراتِ

غربا وشرقا قد أقاموا حربهـــم

وتنـــكّروا لتجــــاور ٍ وصلاتِ

هدموا بأرضِ الله ِ بيتا أكــرما ً

رحم ٌ تعلــق َ دامع ُ الحدقـــــاتِ

البابري ّ بعرش الله متصــــلٌ

وصليب( هتلر) ينفث الحملاتِ

ويهودُ داست أرضَ قدس غيلة ً

وتوعــــدوا الإســلام بالويـلاتِ

قد احرقوا القران جهرا في الضحى

والعرْب ُ في الأعيــادِ والحفلاتِ

يتخطّـفون الناس ً قصدا بائنـــا

قامــت وما فتئت على الخيـراتِ

إلا على الديــنِ القويــمِ جماعـة ٌ

أو نعقُ  بـوم  أو خسيسُ صفاتِ

ما ضرّهم لوم ٌ بدين ِ المصطفى

 

 

       الله أكبرُ يا مــآذن ُ كبّــري منكــم ْ وان ْ عــزّ الدواء ُ أُســاتي

 

 

والفتــح ُ آت ٍ واثــقُ الخطواتِ

نصرٌ عزيزٌ إنْ صدقتــم ْ نيــة ً

بـــدم ٍ عزيز ٍ  ليس بالصدقاتِ

يـــا أمة ‍َ الإسلام ِ بذلا ً خالصا ً

يلـقى نيــوبَ المـوتِ بالقبــلاتِ

تمضون للفردوس مضية  واثقِ ٍ

فيهِ الشفــاءُ  وأصــلُ كلّ ثـباتِ

وتيممــوا إسلامــكم وتمسـكوا

حتى يكــونَ النشءُ خــير بنـــاةِ

ربّوا البيوت على الفضيلةِ والتقى

بالــــعزم ِ قبـــل ندامـــة وفواتِ

والله ُ يقبل توبة ،ً إنْ  ُأخِلصَـت ْ

ويجيء ُ معتصــــم ٌ بغـــيرِ أناةِ

الله أكبرُ ،  سوف َ تعلو صيحـة ًَ

ويعــــود َ للإســـلام ِ حُكـْـمُ حياةِ

ليعيد مجـدَ الديــــنِ سيفــا قاهرا

ويخــاف َ فينا الله  َ حلفُ عصاةِ

لنعــود َ نبراسا ً وهديــاً خالصـا

صدقــــا وأخلصنــا له الدعواتِ

نلقى الإلــــهَ وقد وفيْنــا وعدنـــا
